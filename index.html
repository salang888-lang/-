// Функция расчёта с полным суммированием мяса и специй function calculateBatchCost(ingredients, finishedWeight, markupPct = 40) { let totalCost = 0; let totalRawWeightKg = 0; // 1. Считаем стоимость и переводим абсолютно все ингредиенты в КГ для веса ingredients.forEach(item => { const itemCost = item.weight * item.price; totalCost += itemCost; const unit = item.unit.toLowerCase().trim(); // Перевод единиц измерения в килограммы if (unit === 'кг') { totalRawWeightKg += item.weight; } else if (unit === 'г' || unit === 'гр' || unit === 'грамм') { totalRawWeightKg += (item.weight / 1000); // граммы в кг } // Оболочка (м) и щепа/энергия (партия/шт) на вес самого фарша не влияют, // но их стоимость выше уже плюсуется в totalCost! }); // 2. Если вес готовой продукции ещё не введён if (!finishedWeight || finishedWeight <= 0) { return { totalCost: totalCost.toFixed(2), rawMeatWeight: totalRawWeightKg.toFixed(3), finishedWeight: 0, shrinkagePct: 0, yieldPct: 0, costPerKgFinished: 0 }; } // 3. Расчёт фактической усушки и выходов от ВСЕЙ массы замеса (мясо + специи) const weightLossKg = totalRawWeightKg - finishedWeight; const shrinkagePct = (weightLossKg / totalRawWeightKg) * 100; const yieldPct = (finishedWeight / totalRawWeightKg) * 100; // 4. Себестоимость 1 кг готового продукта и цена продажи const costPerKgFinished = totalCost / finishedWeight; const sellingPricePerKg = costPerKgFinished * (1 + markupPct / 100); const profitPerKg = sellingPricePerKg - costPerKgFinished; const totalProfit = profitPerKg * finishedWeight; return { totalCost: totalCost.toFixed(2), rawMeatWeight: totalRawWeightKg.toFixed(3), // Полный вес сырья (мясо + специи + соль) finishedWeight: finishedWeight.toFixed(3), shrinkagePct: shrinkagePct.toFixed(2), // % усушки yieldPct: yieldPct.toFixed(2), // % выхода costPerKgFinished: costPerKgFinished.toFixed(2), // Себестоимость 1 кг готового sellingPricePerKg: sellingPricePerKg.toFixed(2), // Цена за 1 кг с наценкой totalProfit: totalProfit.toFixed(2) // Чистая прибыль с партии }; }
